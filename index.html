📊 スプレッドシート連携（Google Apps Script）

以下の手順でGASを設定すると、送信データが自動でスプレッドシートに蓄積されます。

1Googleドライブで新しいスプレッドシートを作成し「拡張機能」→「Apps Script」を開く
2下のコードを全て貼り付けて保存（Ctrl+S）する
// こすもすインシデント報告システム - GAS function doPost(e) { try { const data = JSON.parse(e.postData.contents); const ss = SpreadsheetApp.getActiveSpreadsheet(); let sheet = ss.getSheetByName("インシデント一覧"); if (!sheet) { sheet = ss.insertSheet("インシデント一覧"); const h = ["ID","記入日","発生日","発生時刻","所属事業所","報告者名", "利用者様","発生場所","発生分類","原因分類", "発生状況と対応","原因と予防策","所長コメント","登録日時"]; sheet.appendRow(h); sheet.getRange(1,1,1,h.length).setFontWeight("bold") .setBackground("#1a6b4a").setFontColor("#ffffff"); sheet.setFrozenRows(1); } if (data.action === "submit") { sheet.appendRow([ data.id, data.reportDate, data.incidentDate, data.incidentTime, data.office, data.reporter, data.patientName, data.location, data.incType, data.causeType, data.situation, data.prevention, "", new Date().toLocaleString("ja-JP",{timeZone:"Asia/Tokyo"}) ]); } if (data.action === "comment") { const vals = sheet.getDataRange().getValues(); for (let i = 1; i < vals.length; i++) { if (String(vals[i][0]) === String(data.id)) { sheet.getRange(i+1, 13).setValue(data.comment); break; } } } if (data.action === "fetch") { const vals = sheet.getDataRange().getValues(); // Date型を文字列に変換するヘルパー const toStr = (v) => { if (v instanceof Date) { const y = v.getFullYear(); if (y < 2000 || y > 2100) return ''; const m = String(v.getMonth()+1).padStart(2,'0'); const d = String(v.getDate()).padStart(2,'0'); return y+'-'+m+'-'+d; } return v === null || v === undefined ? '' : String(v); }; const toTimeStr = (v) => { if (v instanceof Date) { const h = String(v.getHours()).padStart(2,'0'); const m = String(v.getMinutes()).padStart(2,'0'); return h+':'+m; } return v === null || v === undefined ? '' : String(v); }; const rows = vals.slice(1).map(r => ({ id:r[0], reportDate:toStr(r[1]), incidentDate:toStr(r[2]), incidentTime:toTimeStr(r[3]), office:r[4], reporter:r[5], patientName:r[6], location:r[7], incType:r[8], causeType:r[9], situation:r[10], prevention:r[11], comment:r[12], timestamp:r[13] })); return ContentService.createTextOutput(JSON.stringify({ok:true,data:rows})) .setMimeType(ContentService.MimeType.JSON); } if (data.action === "delete") { const vals = sheet.getDataRange().getValues(); for (let i = 1; i < vals.length; i++) { if (String(vals[i][0]) === String(data.id)) { sheet.deleteRow(i+1); break; } } return ContentService.createTextOutput(JSON.stringify({ok:true})) .setMimeType(ContentService.MimeType.JSON); } return ContentService.createTextOutput(JSON.stringify({ok:true})) .setMimeType(ContentService.MimeType.JSON); } catch(err) { return ContentService.createTextOutput(JSON.stringify({ok:false,error:err.toString()})) .setMimeType(ContentService.MimeType.JSON); } } function doGet(e){ return doPost(e); }
